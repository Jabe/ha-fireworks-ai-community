# Dev-only dependencies for local validation of the integration.
# These are NOT shipped — runtime requirements live in
# custom_components/fireworks_ai_community/manifest.json.

# A real Home Assistant install so imports/types resolve like they do at
# runtime. Min supported version mirrors hacs.json ("homeassistant": "2025.7.0").
homeassistant>=2025.7.0

# Matches the manifest's floor. Deliberately NOT pinned to an exact version:
# Home Assistant ships its own `openai==<x>` line in package_constraints.txt
# (core's openai_conversation uses the same library) and applies that file as a
# pip constraint when installing custom-integration requirements. An exact pin
# here that differs from core's — e.g. ours 2.21.0 vs core 2.45.0 in 2026.8 —
# makes the install unresolvable and setup fails with
# "Requirements for fireworks_ai_community not found". Floor + core's constraint
# means we always run whatever version core itself uses.
openai>=2.21.0

# Used by entity.py for tool/structured-output schema conversion (normally a
# transitive HA dependency; listed explicitly so it is always present).
voluptuous-openapi
